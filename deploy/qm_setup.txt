ls /root/.ssh/id_ed25519.pub || ssh-keygen -t ed25519

qm stop <vmid> && qm destroy <vmid>
qm create <vmid> --name agent --ostype l26 --cores 1 --memory 1024 \
  --scsihw virtio-scsi-single --serial0 socket --agent enabled=1 --net0 virtio,bridge=vmbr0
qm set <vmid> --scsi0 local-lvm:0,import-from=/root/debian-13-genericcloud-amd64.qcow2
qm set <vmid> --ide2 local-lvm:cloudinit --boot order=scsi0
qm set <vmid> --ciuser <user> --sshkeys /root/.ssh/id_ed25519.pub --ipconfig0 ip=<agent-ip>/24,gw=<gateway>
qm resize <vmid> scsi0 15G
qm start <vmid>
